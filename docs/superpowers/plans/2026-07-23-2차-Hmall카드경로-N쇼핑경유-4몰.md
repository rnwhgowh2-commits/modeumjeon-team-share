# [2026-07-23 2차] 카드 즉시할인 목록 — 「구매 가능 가격 > 즉시 할인」 영역.
#   실측(스펙 §11): 카드명+% 가 일자별 로테이션(7/22 KB국민·삼성 → 7/23 삼성·현대 5%).
#   → 크롤 당일 값을 그대로 쓴다(사장님 확정). 없으면 [] (폴백 금지).
HMALL_CARD_PATTERN = re.compile(
    r'class="dc-card[^"]*">\s*([^<]{2,20}?)\s*<strong[^>]*>(\d+)', re.S)


def _parse_hmall_cards(html: str) -> list[dict]:
    """[{label, rate}] — rate 는 퍼센트 단위(5 = 5%). 금액은 사이트가 안 주므로 rate 만."""
    out, seen = [], set()
    for m in HMALL_CARD_PATTERN.finditer(html):
        label = (m.group(1) or '').strip()
        try:
            rate = float(m.group(2))
        except ValueError:
            continue
        if not label or rate <= 0:
            continue
        key = (label, rate)
        if key in seen:
            continue
        seen.add(key)
        out.append({'label': label, 'rate': rate})
    return out
```

// [2026-07-23 2차] Hmall 카드 즉시할인 — 렌더 DOM(일자별 로테이션, 크롤 당일 값 사용)
function hmallCardDiscounts(doc) {
  const out = [], seen = {};
  doc.querySelectorAll('[class*="crdImdDlTagTmp"] .dc-card').forEach((el) => {
    const label = (el.childNodes[0] && el.childNodes[0].textContent || '').trim();
    const st = el.querySelector('strong');
    const rate = st ? parseFloat(st.textContent) : 0;
    const k = label + '|' + rate;
    if (label && rate > 0 && !seen[k]) { seen[k] = 1; out.push({ label, rate }); }
  });
  return out;
}

"""롯데아이몰 플러스쿠폰(네이버 경유 발급형) 수집·매칭.

수집: GET /mypage/searchCouponList.lotte?order_gubun=regi_dt&coupon_type=P&pageIdx=N (15p)
      — 로그인 세션 필요. 쿠폰명에 요율%와 카테고리 접미사가 들어 있다(실측 스펙 §11-5).
매칭: 상품 카테고리(breadcrumb)와 쿠폰 접미사가 **확실히** 대응할 때만 채택.
      애매하면 None = 안 깎음(매입가 과대 = 안전 방향, 사장님 확정 2026-07-23).
"""
import re

_NAME_RE = re.compile(r'(?P<naver>네이버)?[^\d]*(?P<rate>\d+(?:\.\d+)?)%\s*플러스할인쿠폰_?(?P<cat>[가-힣]+)?')

# 쿠폰 접미사 → 상품 카테고리 키워드(확실한 것만. 모르면 넣지 않는다 = 안 깎음)
CATEGORY_MAP = {
    '잡화': ('잡화', '신발', '가방', '슈즈', '스니커즈', '운동화'),
    '의류': ('의류', '패션의류', '티셔츠', '아우터', '팬츠', '원피스'),
}


def parse_coupon_name(name: str) -> dict | None:
    m = _NAME_RE.search(name or '')
    if not m:
        return None
    cat = (m.group('cat') or '').strip()
    cat = re.sub(r'\d+$', '', cat)          # '백화점5' → '백화점'
    return {'is_naver': bool(m.group('naver')), 'rate': float(m.group('rate')),
            'category': cat}


def pick_naver_coupon(coupons: list, product_category: str) -> dict | None:
    """네이버 플러스쿠폰 중 상품 카테고리에 확실히 맞는 최대 요율 1장. 없으면 None."""
    pc = (product_category or '').replace(' ', '')
    if not pc:
        return None
    best = None
    for c in coupons or []:
        info = parse_coupon_name(c.get('name') or '')
        if not info or not info['is_naver'] or info['rate'] <= 0:
            continue
        keys = CATEGORY_MAP.get(info['category'])
        if not keys:
            continue                        # 모르는 접미사 = 안 깎음
        if not any(k in pc for k in keys):
            continue
        if best is None or info['rate'] > best['rate']:
            best = info
    return best
```
