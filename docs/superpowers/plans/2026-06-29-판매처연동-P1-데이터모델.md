        at = Column(DateTime, default=lambda: datetime.now(timezone.utc), nullable=False)
        __table_args__ = (
            Index("ix_cce_lookup", "set_id", "market", "canonical_sku", "field", "at"),

    """[구성 레이어] 판매처/소싱 변동 이벤트 기록 — 값이 바뀔 때만 append."""
    from __future__ import annotations
    from datetime import datetime, timezone
    from sqlalchemy.orm import Session
    from lemouton.sets.models import ChannelChangeEvent

    def record_change(session, *, set_id, market, canonical_sku, field, source,
                      prev_value, next_value, at=None):
        if prev_value == next_value:
            return False
        session.add(ChannelChangeEvent(
            set_id=set_id, market=market, canonical_sku=canonical_sku,
            field=field, source=source, prev_value=prev_value, next_value=next_value,
            at=at or datetime.now(timezone.utc)))
        return True

    """기존 테이블 누락 컬럼 ALTER — create_all 이 못 하는 컬럼 추가를 멱등 처리."""
    from sqlalchemy import inspect, text
    _NEW_COLS = {"mkt_stock":"INTEGER","mkt_price":"INTEGER","mkt_fetched_at":"TIMESTAMP"}
    def ensure_market_columns(engine):
        insp = inspect(engine)
        if "set_channel_options" not in insp.get_table_names():
            return
        existing = {c["name"] for c in insp.get_columns("set_channel_options")}
        with engine.begin() as conn:
            for col, typ in _NEW_COLS.items():
                if col not in existing:
                    conn.execute(text(f"ALTER TABLE set_channel_options ADD COLUMN {col} {typ}"))
