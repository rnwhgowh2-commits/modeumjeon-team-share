## 판별 논리 (본 플랜의 핵심)

계정 A·B, IP1(서버)·IP2(로컬) 4조합으로 **제한의 열쇠(key)** 를 특정한다.

**절차:** A@IP1 로 한도를 소진해 429 를 받은 **직후 3초 이내**에 아래 2개 프로브를 각 1회씩 쏜다.

- **P1** = B@IP1 (다른 계정, 같은 IP) 1회
- **P2** = A@IP2 (같은 계정, 다른 IP) 1회

| P1 결과 | P2 결과 | 결론 | 실무 의미 |
|---|---|---|---|
| 200 | 429 | **계정(키)별** | 계정을 늘리면 총 처리량이 늘어난다 |
| 429 | 200 | **IP별** | 계정을 늘려도 소용없다. IP(서버)를 늘려야 한다 |
| 200 | 200 | **계정 × IP 쌍별** | 둘 다 늘리면 곱으로 늘어난다 |
| 429 | 429 | **판매자/마켓 전역** | 늘릴 방법 없음. 큐로 아껴 써야 한다 |

- **P3(엔드포인트 범위)** = A@IP1 로 엔드포인트 E1 소진 직후 E2 를 1회. 200 이면 **엔드포인트별 버킷**, 429 면 **공통 버킷**.
  - 옥션·G마켓은 문서상 이미 엔드포인트마다 수치가 다르므로(30/분 vs 5초당1) 엔드포인트별 버킷일 확률이 높다 — 그래도 실측으로 확인한다.
- **P4(윈도우 형태)** = 429 직후 1초 간격 단발 호출을 60초간 반복하며 첫 200 시각 기록.
  - 회복이 **분 경계(:00초)에 몰리면 고정창**, **일정 시간 후 점진 회복이면 토큰버킷/슬라이딩**.

**판정 불가 처리:** P1/P2 응답이 429 가 아닌 다른 코드(403·401·5xx)면 `판별불가`. 429 가 아닌 것을 429 로 간주하지 않는다.

## 수치 측정 논리 (x초에 y개)

엔드포인트마다 3개 값을 구한다.

1. **버스트 용량 C** — 120초 완전 휴식 후, 간격 없이 연속 호출. 첫 429 직전까지의 성공 수 = C.
2. **지속 속도 R** — 이분탐색. 후보 r 에 대해 `r req/s` 로 30초 유지 → 429 0건이면 통과. 통과하는 최대 r = R. 탐색 범위 0.2 ~ 20 req/s, 3회 반복.
3. **회복 시간 T** — P4 결과.

최종 표기는 **"C개 버스트 후 R req/s 지속"** 형태로 적는다. `분당 N회` 로 뭉개지 않는다.

```python
# scripts/ratelimit_probe/config.py
# -*- coding: utf-8 -*-
"""프로브 안전장치 — 조회계 화이트리스트·예산·킬스위치.

여기 등재되지 않은 경로는 절대 호출하지 않는다. 등록·수정·삭제는 어떤 경우에도 금지.
"""
from __future__ import annotations

import os
from dataclasses import dataclass
from pathlib import Path

STOP_FILE = Path(__file__).with_name("STOP")
DEFAULT_BUDGET = 2000            # 마켓당 총 호출 상한
CONSECUTIVE_429_ABORT = 20       # 429 연속 이 횟수면 중단
MARKET_GAP_SEC = 300             # 마켓 전환 간격
BURST_REST_SEC = 120             # 버스트 측정 전 휴식


class ProbeForbidden(RuntimeError):
    """화이트리스트 밖 호출 시도."""


@dataclass(frozen=True)
class SafeEndpoint:
    key: str          # "E1" / "E2"
    method: str       # "GET" | "POST"(조회용만)
    path: str
    note: str


# 전부 '조회'만. 파라미터는 존재하지 않는 값을 넣어 데이터 변경 위험을 0 으로 만든다.
SAFE_ENDPOINTS: dict[str, list[SafeEndpoint]] = {
    "coupang": [
        SafeEndpoint("E1", "GET",
                     "/v2/providers/seller_api/apis/api/v1/marketplace/meta/category-related-metas/display-category-codes/{cat}",
                     "카테고리 메타 조회 — 읽기 전용, 판매자 데이터 무관"),
```

```python
# scripts/ratelimit_probe/recorder.py
# -*- coding: utf-8 -*-
"""프로브 호출 기록기 — 시각·상태·응답헤더 전부를 JSONL 로 남긴다.

헤더를 통째로 남기는 이유: 어떤 마켓이 어떤 이름으로 잔여량을 알려주는지 사전에 모른다.
시크릿이 섞일 수 있는 헤더만 마스킹한다.
"""
from __future__ import annotations

import json
import time
from pathlib import Path

_SECRET_HEADERS = {
    "authorization", "openapikey", "x-api-key", "cookie", "set-cookie",
    "api-key", "apikey", "x-auth-token",
}


class Recorder:
    def __init__(self, out_path: Path, *, market: str, account: str, ip_label: str):
        self._path = Path(out_path)
        self._path.parent.mkdir(parents=True, exist_ok=True)
        self._market = market
        self._account = account
        self._ip_label = ip_label

    def record(self, method: str, url: str, resp, *, elapsed_ms: float) -> None:
        row = {
            "ts": time.time(),
            "market": self._market,
            "account": self._account,
            "ip_label": self._ip_label,
            "method": method,
            "url": url,
            "status": getattr(resp, "status_code", None),
            "elapsed_ms": round(float(elapsed_ms), 2),
            "headers": self._clean(getattr(resp, "headers", {}) or {}),
            "body_head": (getattr(resp, "text", "") or "")[:200],
        }
        with self._path.open("a", encoding="utf-8") as f:
            f.write(json.dumps(row, ensure_ascii=False) + "\n")

    @staticmethod
    def _clean(headers) -> dict:
        out = {}
        for k, v in dict(headers).items():
            out[k] = "***REDACTED***" if k.lower() in _SECRET_HEADERS else v
        return out
```

```python
# -*- coding: utf-8 -*-
"""단발 프로브 — 기존 클라이언트의 인증만 재사용하고 리미터·재시도는 끈다.

재시도가 켜져 있으면 429 를 내부에서 삼켜 측정이 불가능하므로 max_retries=1 로 고정.
"""
from __future__ import annotations

import time
from dataclasses import dataclass
from typing import Optional

from .config import SAFE_ENDPOINTS, assert_safe, ProbeForbidden
from .recorder import Recorder


@dataclass(frozen=True)
class ProbeResult:
    status: Optional[int]
    is_rate_limited: bool
    retry_after: Optional[int]
    elapsed_ms: float
    error: Optional[str] = None


def probe_config_overrides(market: str) -> dict:
    """리미터·재시도를 무력화하는 설정 오버라이드."""
    ov = {"max_retries": 1, "retry_backoff_sec": 0, "rate_limit_per_sec": 10**6}
    if market in ("auction", "gmarket"):
        ov["order_min_interval_sec"] = 0
    return ov


def _endpoint(market: str, key: str):
    for e in SAFE_ENDPOINTS.get(market, []):
        if e.key == key:
            return e
    raise ProbeForbidden(f"알 수 없는 엔드포인트 키: {market} {key}")


def _raw_request(method: str, url: str, headers: dict, body: Optional[dict], timeout: float):
    """실제 HTTP. 테스트에서 monkeypatch 대상."""
    import requests
    if method == "GET":
        return requests.get(url, headers=headers, timeout=timeout)
    return requests.post(url, json=body or {}, headers=headers, timeout=timeout)


def build_auth_headers(market: str, account: str) -> tuple[str, dict]:
    """마켓별 base_url·인증 헤더. 기존 클라이언트 인증 로직 재사용."""
    from lemouton.auth import secrets as S
    from shared import platforms as P

    ov = probe_config_overrides(market)
    if market == "coupang":
        from shared.platforms.coupang.client import CoupangClient
        c = S.load_credentials(market="coupang", env_prefix=account)
        cl = CoupangClient(config={**P.COUPANG, "access_key": c.access_key,
                                   "secret_key": c.secret_key,
                                   "vendor_id": c.vendor_id, **ov})
        return cl.base_url, cl._headers_for("GET", "")   # HMAC 은 경로별 — 아래 주석 참조
    if market == "lotteon":
        from shared.platforms.lotteon.client import LotteonClient
        c = S.load_credentials(market="lotteon", env_prefix=account)
        cl = LotteonClient(config={**P.LOTTEON, "api_key": c.api_key,
                                   "tr_no": c.tr_no, **ov})
        return cl.base_url, cl._headers()
    if market == "eleven11":
        from shared.platforms.eleven11.client import Eleven11Client
        c = S.load_credentials(market="eleven11", env_prefix=account)
        cl = Eleven11Client(config={**P.ELEVEN11, "openapi_key": c.openapi_key, **ov})
        return cl.base_url, cl._headers()
    if market == "smartstore":
        from shared.platforms.smartstore.client import SmartStoreClient
        c = S.load_credentials(market="smartstore", env_prefix=account)
        cl = SmartStoreClient(config={**P.SMARTSTORE, "client_id": c.client_id,
                                      "client_secret": c.client_secret, **ov})
        return cl.base_url, cl._auth_headers()
    if market in ("auction", "gmarket"):
        from shared.platforms.esm.client import EsmClient
        base = P.AUCTION if market == "auction" else P.GMARKET
        c = S.load_credentials(market=market, env_prefix=account)
        cl = EsmClient(config={**base, "master_id": c.master_id,
                               "secret_key": c.secret_key,
                               "seller_id": c.seller_id, **ov})
        return cl.base_url, cl._headers()
    raise ProbeForbidden(f"미지원 마켓: {market}")


def probe_once(market: str, endpoint_key: str, *, account: str,
               recorder: Recorder, ip_label: str = "IP1",
               timeout: float = 15.0) -> ProbeResult:
    """단발 호출 1회. 429 는 예외가 아니라 값으로 돌려준다."""
    ep = _endpoint(market, endpoint_key)
    assert_safe(market, ep.method, ep.path)

    url = ep.path if ep.path.startswith("http") else None
    headers: dict = {}
    if url is None:
        base, headers = build_auth_headers(market, account)
        url = base + ep.path
    else:
        _, headers = build_auth_headers(market, account)

    t0 = time.monotonic()
    try:
        resp = _raw_request(ep.method, url, headers, {}, timeout)
    except Exception as e:  # noqa: BLE001 — 네트워크 실패도 데이터로 남긴다
        return ProbeResult(None, False, None, (time.monotonic() - t0) * 1000, str(e))
    elapsed = (time.monotonic() - t0) * 1000
    recorder.record(ep.method, url, resp, elapsed_ms=elapsed)

    status = resp.status_code
    ra = None
    if status == 429:
        try:
            ra = int(resp.headers.get("Retry-After", "")) if resp.headers.get("Retry-After") else None
        except ValueError:
            ra = None
    return ProbeResult(status, status == 429, ra, elapsed)
```

```python
    """안전장치 발동으로 프로브 중단."""

    """호출 1건마다 tick() → 응답마다 observe(status). 위반 시 즉시 ProbeAborted."""

    def __init__(self, *, market: str, budget: int,
                 max_consecutive_429: int = None):
        from .config import budget_for, CONSECUTIVE_429_ABORT
        self.market = market
        self.budget = int(budget if budget is not None else budget_for(market))
        self.max_consecutive_429 = int(max_consecutive_429 or CONSECUTIVE_429_ABORT)
        self.used = 0
        self.consecutive_429 = 0

    def tick(self) -> None:
        from . import config
        if config.STOP_FILE.exists():
            raise ProbeAborted(f"[{self.market}] 킬스위치 파일 감지 — 중단")
        if self.consecutive_429 >= self.max_consecutive_429:
            raise ProbeAborted(
                f"[{self.market}] 429 연속 {self.consecutive_429}회 — 중단")
        if self.used >= self.budget:
            raise ProbeAborted(
                f"[{self.market}] 호출 예산 {self.budget} 소진 — 중단")
        self.used += 1

    def observe(self, status) -> None:
        if status == 429:
            self.consecutive_429 += 1
        else:
            self.consecutive_429 = 0
```

```python
# scripts/ratelimit_probe/measure.py
# -*- coding: utf-8 -*-
"""수치 측정 — 버스트 용량 C, 지속 속도 R, 회복 시간 T.

결과 표기는 "C개 버스트 후 R req/s 지속". '분당 N회' 로 뭉개지 않는다.
"""
from __future__ import annotations

import time
from dataclasses import dataclass
from typing import Callable, Optional


@dataclass
class BurstResult:
    capacity: int
    hit_429: bool
    note: str = ""


def measure_burst(probe: Callable[[], "ProbeResult"], *,   # noqa: F821
                  max_calls: int = 200, rest_sec: float = 120.0) -> BurstResult:
    """완전 휴식 후 간격 없이 연속 호출 → 첫 429 직전까지의 성공 수 = 버스트 용량."""
    if rest_sec:
        time.sleep(rest_sec)
    ok = 0
    for _ in range(max_calls):
        r = probe()
        if r.is_rate_limited:
            return BurstResult(ok, True)
        if r.status is None or r.status >= 400:
            return BurstResult(ok, False,
                               f"429 아닌 오류 status={r.status} — 측정 중단(판별불가)")
        ok += 1
    return BurstResult(ok, False, "상한 미도달 — max_calls 를 올려 재측정 필요")


def search_sustained_rate(holds_at: Callable[[float], bool], *,
                          lo: float = 0.2, hi: float = 20.0,
                          rounds: int = 8) -> float:
    """holds_at(rate) 가 True = 그 속도로 30초 유지해도 429 없음. 통과하는 최대 rate 반환."""
    if not holds_at(lo):
        return 0.0
    if holds_at(hi):
        return hi
    for _ in range(rounds):
        mid = (lo + hi) / 2
        if holds_at(mid):
            lo = mid
        else:
            hi = mid
    return round(lo, 2)


def make_hold_check(probe: Callable[[], "ProbeResult"], *,  # noqa: F821
                    hold_sec: float = 30.0,
                    cooldown_sec: float = 60.0) -> Callable[[float], bool]:
    """rate req/s 로 hold_sec 유지해 429 가 0건이면 True. 매 시도 후 쿨다운."""
    def holds_at(rate: float) -> bool:
        interval = 1.0 / rate
        deadline = time.monotonic() + hold_sec
        while time.monotonic() < deadline:
            r = probe()
            if r.is_rate_limited:
                time.sleep(cooldown_sec)
                return False
            time.sleep(interval)
        time.sleep(cooldown_sec)
        return True
    return holds_at


def classify_recovery(*, t_429: float, t_first_200: float,
                      tolerance_sec: float = 2.0) -> str:
    """회복 시각이 분 경계에 붙으면 고정창, 아니면 토큰버킷/슬라이딩."""
    sec_in_min = t_first_200 % 60.0
    near_boundary = sec_in_min <= tolerance_sec or sec_in_min >= (60.0 - tolerance_sec)
    if near_boundary and (t_first_200 - t_429) > tolerance_sec:
        return "고정창(분 경계)"
    return "토큰버킷/슬라이딩"


def measure_recovery(probe: Callable[[], "ProbeResult"], *,   # noqa: F821
                     t_429: float, max_wait_sec: float = 120.0) -> Optional[float]:
    """429 직후 1초 간격 단발 호출 → 첫 200 시각 반환(못 찾으면 None)."""
    deadline = time.monotonic() + max_wait_sec
    while time.monotonic() < deadline:
        r = probe()
        if r.status == 200:
            return time.time()
        time.sleep(1.0)
    return None
```

```python
# scripts/ratelimit_probe/discriminate.py
# -*- coding: utf-8 -*-
"""제한 단위 판별 — 계정별 / IP별 / 계정×IP / 전역.

A@IP1 을 429 까지 소진한 직후 3초 이내에
  P1 = B@IP1 (다른 계정, 같은 IP)
  P2 = A@IP2 (같은 계정, 다른 IP)
를 각 1회 쏜 결과로 판정한다. 429 가 아닌 응답은 절대 429 로 간주하지 않는다.
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Optional


@dataclass(frozen=True)
class LimitKeyVerdict:
    verdict: str
    reason: str


_TABLE = {
    (False, True): ("계정(키)별", "다른 계정은 통과, 같은 계정은 다른 IP에서도 막힘"),
    (True, False): ("IP별", "다른 계정도 막힘, 같은 계정이 다른 IP에서는 통과"),
    (False, False): ("계정×IP 쌍별", "다른 계정도 다른 IP도 모두 통과"),
    (True, True): ("판매자/마켓 전역", "다른 계정도 다른 IP도 모두 막힘"),
}


def classify_limit_key(*, p1_status: Optional[int],
                       p2_status: Optional[int]) -> LimitKeyVerdict:
    if p1_status is None:
        return LimitKeyVerdict("판별불가", "P1 미실행 — 두 번째 계정이 없음")
    if p2_status is None:
        return LimitKeyVerdict("판별불가", "P2 미실행 — 두 번째 IP 미확보")
    for label, st in (("P1", p1_status), ("P2", p2_status)):
        if st not in (200, 429):
            return LimitKeyVerdict(
                "판별불가",
                f"{label} 응답이 {st} — 429/200 이 아니라 판정 불가"
                f"{' (IP 미등록 의심)' if st == 403 else ''}")
    key = (p1_status == 429, p2_status == 429)
    verdict, reason = _TABLE[key]
    return LimitKeyVerdict(verdict, reason)


def classify_endpoint_scope(*, e2_status: Optional[int]) -> str:
    """E1 소진 직후 E2 호출 결과 → 버킷이 엔드포인트별인지 공통인지."""
    if e2_status == 200:
        return "엔드포인트별 버킷"
    if e2_status == 429:
        return "공통 버킷"
    return "판별불가"
```

```python
# -*- coding: utf-8 -*-
"""프로브 CLI.

    python -m scripts.ratelimit_probe.run --market lotteon --phase smoke  --account LOTTEON_MAIN
    python -m scripts.ratelimit_probe.run --market lotteon --phase burst  --account LOTTEON_MAIN
    python -m scripts.ratelimit_probe.run --market lotteon --phase rate   --account LOTTEON_MAIN
    python -m scripts.ratelimit_probe.run --market lotteon --phase discriminate \
        --account LOTTEON_MAIN --account-b LOTTEON_SUB --ip-label IP1
"""
from __future__ import annotations

import argparse
import sys
from pathlib import Path

from .config import SAFE_ENDPOINTS, budget_for
from .recorder import Recorder
from . import prober, measure

PHASES = ("smoke", "burst", "rate", "recovery", "discriminate")


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(prog="ratelimit_probe")
    ap.add_argument("--market", required=True)
    ap.add_argument("--phase", required=True, choices=PHASES)
    ap.add_argument("--account", required=True, help="env_prefix (계정 A)")
    ap.add_argument("--account-b", default=None, help="env_prefix (계정 B) — discriminate 용")
    ap.add_argument("--ip-label", default="IP1")
    ap.add_argument("--endpoint", default="E1")
    ap.add_argument("--out", default="scripts/ratelimit_probe/out/probe.jsonl")
    ap.add_argument("--budget", type=int, default=None)
    ap.add_argument("--dry-run", action="store_true")
    a = ap.parse_args(argv)

    if a.market not in SAFE_ENDPOINTS:
        print(f"[오류] 미등록 마켓: {a.market}", file=sys.stderr)
        return 2

    rec = Recorder(Path(a.out), market=a.market, account=a.account,
                   ip_label=a.ip_label)
    guard = prober.ProbeGuard(market=a.market,
                              budget=a.budget or budget_for(a.market))

    def probe(endpoint_key: str = a.endpoint, account: str = a.account):
        guard.tick()
        r = prober.probe_once(a.market, endpoint_key, account=account,
                              recorder=rec, ip_label=a.ip_label)
        guard.observe(r.status)
        return r

    if a.dry_run:
        for ep in SAFE_ENDPOINTS[a.market]:
            print(f"[드라이런] {a.market} {ep.key} {ep.method} {ep.path} — {ep.note}")
        print(f"[드라이런] phase={a.phase} budget={guard.budget} 실호출 없음")
        return 0

    if a.phase == "smoke":
        for ep in SAFE_ENDPOINTS[a.market]:
            r = probe(ep.key)
            print(f"{a.market} {ep.key} → {r.status}")
        return 0

    if a.phase == "burst":
        res = measure.measure_burst(lambda: probe())
        print(f"{a.market} {a.endpoint} 버스트 용량 = {res.capacity} "
              f"(429도달={res.hit_429}) {res.note}")
        return 0

    if a.phase == "rate":
        holds = measure.make_hold_check(lambda: probe())
        r = measure.search_sustained_rate(holds)
        print(f"{a.market} {a.endpoint} 지속 가능 속도 = {r} req/s")
        return 0

    if a.phase == "recovery":
        import time as _t
        while True:
            res = probe()
            if res.is_rate_limited:
                t429 = _t.time()
                break
        t200 = measure.measure_recovery(lambda: probe(), t_429=t429)
        if t200 is None:
            print(f"{a.market} 회복 미관측(120초 내)")
            return 0
        shape = measure.classify_recovery(t_429=t429, t_first_200=t200)
        print(f"{a.market} 회복까지 {t200 - t429:.1f}초 · 형태={shape}")
        return 0

    if a.phase == "discriminate":
        from .discriminate import classify_limit_key, classify_endpoint_scope
        if not a.account_b:
            print("[오류] --account-b 없음 → P1 불가 (판별불가)", file=sys.stderr)
            return 2
        # A@IP1 소진
        while True:
            res = probe("E1", a.account)
            if res.is_rate_limited:
                break
        # 소진 직후 3초 이내에 P1·P3
        p1 = probe("E1", a.account_b).status
        e2 = probe("E2", a.account).status
        print(f"P1(다른계정 같은IP) = {p1}")
        print(f"P3(같은계정 다른엔드포인트) = {e2} → {classify_endpoint_scope(e2_status=e2)}")
        print("P2(같은계정 다른IP) 는 IP2 호스트에서 --ip-label IP2 로 별도 실행 후 합치기")
        v = classify_limit_key(p1_status=p1, p2_status=None)
        print(f"현재 판정: {v.verdict} — {v.reason}")
        return 0

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
# shared/platforms/shared_limiter.py
# -*- coding: utf-8 -*-
"""마켓 단위 공유 토큰버킷.

제한이 IP/전역 단위인 마켓은 계정마다 버킷을 따로 두면 총량이 계정 수만큼 곱해진다.
(예: 롯데온 7계정 × 5 req/s = 35 req/s → 한도 초과)
프로세스 내 마켓당 1개 버킷을 공유한다.

한계: 프로세스가 여러 개면(워커 다중화) 여전히 곱해진다. 그 경우는
      Redis 등 프로세스 간 공유가 필요 — 현재는 단일 프로세스 전제.
"""
from __future__ import annotations

import threading
import time

_LOCK = threading.Lock()
_BUCKETS: dict[str, "_SharedTokenBucket"] = {}


class _SharedTokenBucket:
    def __init__(self, rate_per_sec: float):
        self._rate = float(rate_per_sec)
        self._capacity = float(rate_per_sec)
        self._tokens = float(rate_per_sec)
        self._last = time.monotonic()
        self._lock = threading.Lock()

    def acquire(self) -> None:
        while True:
            with self._lock:
                now = time.monotonic()
                elapsed = now - self._last
                self._last = now
                self._tokens = min(self._capacity, self._tokens + elapsed * self._rate)
                if self._tokens >= 1.0:
                    self._tokens -= 1.0
                    return
                wait = (1.0 - self._tokens) / self._rate
            time.sleep(wait)

    def set_rate_from_replenish(self, rate: float) -> None:
        with self._lock:
            self._rate = max(0.1, float(rate))
            self._capacity = self._rate


def get_limiter(market: str, *, rate_per_sec: float) -> _SharedTokenBucket:
    with _LOCK:
        b = _BUCKETS.get(market)
        if b is None:
            b = _SharedTokenBucket(rate_per_sec)
            _BUCKETS[market] = b
        return b


def _reset_for_test() -> None:
    with _LOCK:
        _BUCKETS.clear()
```
