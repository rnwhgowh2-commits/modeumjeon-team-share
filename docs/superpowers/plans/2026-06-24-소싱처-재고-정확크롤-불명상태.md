# 소싱처 재고 정확 크롤 — "불명(unknown)" 상태 도입 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 재고를 못 읽었을 때 "있음(999)"으로 둔갑시키지 말고 "불명(-1) → ⚠️확인필요 + 수량0 취급"으로 정직하게 표시·차단한다.

**Architecture:** 정수 재고 칸에 센티넬 `-1`(불명)을 추가한다. 종착지 `_resolve_stock`(서버) 한 곳이 `-1`을 "⚠️확인필요·판매제외"로 해석하면 매트릭스·최저가·업로드에 자동 반영된다. 각 소싱처 추출기(확장)는 "애매하면 999" 기본값을 "애매하면 -1"로 뒤집어, `999`는 명시적 in-stock 신호가 있을 때만 부여한다.

**Tech Stack:** Python(Flask, pytest) 서버 = `origin/main`; JavaScript 크롬 확장(MV3 service worker) = 데스크톱 로드 사본 + 레포 사본; node로 추출기 순수함수 회귀 테스트.

**설계 문서:** `docs/superpowers/specs/2026-06-24-소싱처-재고-정확크롤-불명상태-design.md`

```javascript
/** 재고 '불명(-1)' 추출 규칙 회귀 테스트. node test_stock_unknown_extractors.js */

// 무신사 stock 규칙(추출판): 호출실패/미매칭→-1, 품절→0, 수량→N, 그 외 충분→999.
function musinsaStock(invOk, inv) {
  return !invOk ? -1
    : !inv ? -1
    : inv.outOfStock ? 0
    : (typeof inv.remainQuantity === "number" ? Math.max(0, inv.remainQuantity) : 999);
}
// 롯데온 DOM 폴백 stock(추출판): 매칭 N / 텍스트품절0 / liSold0 / 그 외 -1.
function lotteonDomStock(t, liSold) {
  const mm = t.match(/(\d+)\s*개\s*남음/) || t.match(/마지막\s*(\d+)\s*개/);
  return mm ? Math.max(0, parseInt(mm[1], 10))
    : (/품절|일시품절/.test(t) ? 0 : (liSold ? 0 : -1));
}

let fail = 0;
const eq = (label, got, want) => {
  const ok = got === want;
  console.log(`${ok ? "PASS" : "FAIL"}  ${label}`);
  if (!ok) { console.log(`   got ${got} want ${want}`); fail++; }
};

// 무신사
eq("무신사 API 호출실패 → -1 불명", musinsaStock(false, null), -1);
eq("무신사 옵션 미매칭 → -1 불명(999 둔갑 제거)", musinsaStock(true, undefined), -1);
eq("무신사 품절 → 0", musinsaStock(true, { outOfStock: true }), 0);
eq("무신사 한정 5 → 5", musinsaStock(true, { outOfStock: false, remainQuantity: 5 }), 5);
eq("무신사 충분(수량 null) → 999", musinsaStock(true, { outOfStock: false, remainQuantity: null }), 999);

// 롯데온 DOM 폴백
eq("롯데온 DOM '3개 남음' → 3", lotteonDomStock("3개 남음", false), 3);
eq("롯데온 DOM '마지막 2개' → 2", lotteonDomStock("마지막 2개", false), 2);
eq("롯데온 DOM 품절 → 0", lotteonDomStock("품절", false), 0);
eq("롯데온 DOM liSold → 0", lotteonDomStock("", true), 0);
eq("롯데온 DOM 신호없음 → -1 불명(999 둔갑 제거)", lotteonDomStock("", false), -1);

console.log(fail === 0 ? "\nALL PASS" : `\n${fail} FAILED`);
process.exit(fail === 0 ? 0 : 1);
```
