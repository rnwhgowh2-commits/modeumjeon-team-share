```python
# -*- coding: utf-8 -*-
"""재크롤 없는 드라이런 전수 검증 리포트.

이미 저장된 크롤 데이터 + 마진 설정으로 각 마켓에 전송될 값(old→new)을 실제
어댑터 호출 없이 산출한다. full_cycle()의 Phase A(실크롤)는 절대 호출하지 않는다.

실행:
    cd 프로그램/_시스템
    ENVIRONMENT=team-share-dev DISABLE_AUTH=1 python -m scripts.verify_pipeline_dryrun
"""
from __future__ import annotations

from collections import defaultdict


def summarize_uploads(uploads: list[dict]) -> dict:
    """_extract_uploads 결과를 마켓별로 집계하고 이상치를 표면화(순수 함수)."""
    by_market: dict[str, dict] = {}
    zero_price: list[dict] = []
    zero_stock: list[dict] = []
    for u in uploads:
        m = u["market"]
        slot = by_market.setdefault(m, {"count": 0, "items": []})
        slot["count"] += 1
        slot["items"].append(u)
        if not u.get("new_price"):      # 0 또는 None → 전송 금지 후보
            zero_price.append({"market": m, "canonical_sku": u["canonical_sku"]})
        if not u.get("new_stock"):
            zero_stock.append({"market": m, "canonical_sku": u["canonical_sku"]})
    return {
        "total": len(uploads),
        "by_market": by_market,
        "anomalies": {"zero_price": zero_price, "zero_stock": zero_stock},
    }


def build_c_output(session) -> dict:
    """Phase B(pricing) → C(formatter) 만 재사용해 마켓별 페이로드 산출(재크롤 없음).

    a_output(옵션 aggregate)은 저장된 SourceOption 기준으로 만든다 —
    jobs.full_cycle 의 Phase A(fetch_unique_sources=실크롤)는 호출하지 않는다.
    """
    from lemouton.sourcing.pipeline import run_pipeline
    from lemouton.pricing.engine import run_pricing_engine
    from lemouton.pricing.settings import get_pricing_settings
    from lemouton.formatter.pipeline import run_formatter

    # crawlers 없이 매핑/aggregate 만 — 저장된 값으로 a_output 구성(실크롤 배제)
    a_output = run_pipeline(session, crawlers=None) or {}
    settings = get_pricing_settings(session)
    b_output = run_pricing_engine(a_output, settings) or {}
    c_output = run_formatter(session, a_output, b_output) or {}
    return c_output


def main() -> None:
    from shared.db import SessionLocal
    from lemouton.uploader.orchestrator import _extract_uploads
    from lemouton.uploader.runtime import build_sku_by_option

    s = SessionLocal()
    try:
        c_output = build_c_output(s)
        sku_by_option = build_sku_by_option(s)
        uploads = _extract_uploads(c_output, sku_by_option)
        rep = summarize_uploads(uploads)
    finally:
        s.close()

    print(f"총 전송 대상 옵션: {rep['total']}")
    for m, slot in sorted(rep["by_market"].items()):
        print(f"  · {m}: {slot['count']}건")
    zp, zs = rep["anomalies"]["zero_price"], rep["anomalies"]["zero_stock"]
    print(f"⚠️ 0원 전송 후보: {len(zp)}  (전송 금지 대상 — price_guard 가 abort)")
    print(f"⚠️ 재고 0 후보: {len(zs)}")
    for row in zp[:20]:
        print(f"    zero_price {row['market']} {row['canonical_sku']}")
    alerts = c_output.get("alerts", [])
    print(f"알림(매핑 실패 등): {len(alerts)}")
    for a in alerts[:20]:
        print(f"    {a.get('type')} {a.get('model_code') or a.get('canonical_sku')} — {a.get('message')}")


if __name__ == "__main__":
    main()
```

**주의:** `run_pipeline(session, crawlers=None)` 시그니처가 다르면(예: crawlers 필수) 구현 Step 에서 `프로그램/_시스템/lemouton/sourcing/pipeline.py` 의 `run_pipeline` 정의를 읽어, **실크롤을 트리거하지 않고 저장된 옵션으로 aggregate 만** 하는 호출 형태로 맞춘다. `get_pricing_settings` 이름이 다르면 `jobs.py` Phase B 가 쓰는 실제 함수/변수명을 그대로 따른다(정본 = jobs.full_cycle Phase B).
